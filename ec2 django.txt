sudo apt update
sudo apt install python3-pip python3-dev nginx
sudo pip3 install virtualenv
sudo apt install python3-virtualenv
git clone https://github.com/syedmehm00d/softtech.git
cd ~/softtech

virtualenv env
source env/bin/activate
pip install -r requirements.txt
pip install django gunicorn
python manage.py makemigrations
python manage.py migrate
python manage.py collectstatic


deactivate

sudo nano /etc/systemd/system/gunicorn.socket

//code for file gunicorn.socket

[Unit]
Description=gunicorn socket
[Socket]
ListenStream=/run/gunicorn.sock
[Install]
WantedBy=sockets.target


sudo nano /etc/systemd/system/gunicorn.service

//code for gunicorn.service

[Unit]
Description=gunicorn daemon
Requires=gunicorn.socket
After=network.target
[Service]
User=ubuntu
Group=www-data
WorkingDirectory=/home/ubuntu/softtech
ExecStart=/home/ubuntu/softtech/env/bin/gunicorn \
          --access-logfile - \
          --workers 3 \
          --bind unix:/run/gunicorn.sock \
          softtech.wsgi:application
[Install]
WantedBy=multi-user.target




sudo systemctl start gunicorn.socket
sudo systemctl enable gunicorn.socket


sudo nano /etc/nginx/sites-available/blog

// code for blog file

server {
    listen 80 default_server;
    server_name _;
    location = /favicon.ico { access_log off; log_not_found off; }
    location /static/ {
        root /home/ubuntu/softtech;
    }

    location /media/ {
        root /home/ubuntu/softtech;
    }
    location / {
        include proxy_params;
        proxy_pass http://unix:/run/gunicorn.sock;
    }
}


sudo ln -s /etc/nginx/sites-available/blog /etc/nginx/sites-enabled/



sudo gpasswd -a www-data username

sudo chown -R www-data:www-data /home/ubuntu/softtech/media
sudo chmod -R 755 /home/ubuntu/softtech/media

sudo chown -R ubuntu:ubuntu /home/ubuntu/softtech/media/
sudo chmod -R 777 /home/ubuntu/softtech/media/

//remove porit listing 80 from file /etc/nginx/sites-available/default
sudo systemctl restart gunicorn

sudo nginx -t
